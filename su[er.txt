-- =========================================================

CREATE TABLE CUSTOMER (
    CUSTOMER_ID   NUMBER(6)      PRIMARY KEY,
    FIRST_NAME    VARCHAR2(30)   NOT NULL,
    SURNAME       VARCHAR2(30)   NOT NULL,
    ADDRESS       VARCHAR2(60),
    PHONE_NUM     VARCHAR2(15),
    EMAIL         VARCHAR2(60)
);

CREATE TABLE STAFF (
    STAFF_ID      NUMBER(6)      PRIMARY KEY,
    FIRST_NAME    VARCHAR2(30)   NOT NULL,
    SURNAME       VARCHAR2(30)   NOT NULL,
    POSITION      VARCHAR2(30),
    PHONE_NUM     VARCHAR2(15),
    ADDRESS       VARCHAR2(60),
    EMAIL         VARCHAR2(60)
);

CREATE TABLE DRIVER (
    DRIVER_ID     NUMBER(6)      PRIMARY KEY,
    FIRST_NAME    VARCHAR2(30)   NOT NULL,
    SURNAME       VARCHAR2(30)   NOT NULL,
    DRIVER_CODE   VARCHAR2(10),
    PHONE_NUM     VARCHAR2(15),
    ADDRESS       VARCHAR2(60)
);

CREATE TABLE VEHICLE (
    VIN_NUMBER    VARCHAR2(20)   PRIMARY KEY,
    VEHICLE_TYPE  VARCHAR2(30),
    MILEAGE       NUMBER(10),
    COLOUR        VARCHAR2(20),
    MANUFACTURER  VARCHAR2(30)
);

-- =========================================================
-- Dependent tables
-- =========================================================

CREATE TABLE BILLING (
    BILL_ID       NUMBER(6)      PRIMARY KEY,
    CUSTOMER_ID   NUMBER(6)      NOT NULL,
    STAFF_ID      NUMBER(6)      NOT NULL,
    BILL_DATE     DATE,
    CONSTRAINT FK_BILLING_CUSTOMER FOREIGN KEY (CUSTOMER_ID) REFERENCES CUSTOMER(CUSTOMER_ID),
    CONSTRAINT FK_BILLING_STAFF    FOREIGN KEY (STAFF_ID)    REFERENCES STAFF(STAFF_ID)
);

CREATE TABLE DELIVERY_ITEM (
    ITEM_ID       NUMBER(6)      PRIMARY KEY,
    DESCRIPTION   VARCHAR2(100),
    STAFF_ID      NUMBER(6)      NOT NULL,
    CONSTRAINT FK_ITEM_STAFF FOREIGN KEY (STAFF_ID) REFERENCES STAFF(STAFF_ID)
);

CREATE TABLE DRIVER_DELIVERIES (
    DRIVER_DELIVERY_ID  NUMBER(6)     PRIMARY KEY,
    VIN_NUMBER          VARCHAR2(20)  NOT NULL,
    DRIVER_ID           NUMBER(6)     NOT NULL,
    ITEM_ID             NUMBER(6)     NOT NULL,
    CONSTRAINT FK_DD_VEHICLE FOREIGN KEY (VIN_NUMBER) REFERENCES VEHICLE(VIN_NUMBER),
    CONSTRAINT FK_DD_DRIVER  FOREIGN KEY (DRIVER_ID)  REFERENCES DRIVER(DRIVER_ID),
    CONSTRAINT FK_DD_ITEM    FOREIGN KEY (ITEM_ID)    REFERENCES DELIVERY_ITEM(ITEM_ID)
);

SET SERVEROUTPUT ON; DECLARE v_staff_id STAFF.STAFF_ID%TYPE; v_first_name STAFF.FIRST_NAME%TYPE; v_surname STAFF.SURNAME%TYPE; v_delivery_cnt NUMBER; BEGIN SELECT S.STAFF_ID, S.FIRST_NAME, S.SURNAME, COUNT(*) INTO v_staff_id, v_first_name, v_surname, v_delivery_cnt FROM STAFF S JOIN DELIVERY_ITEM DI ON S.STAFF_ID = DI.STAFF_ID JOIN DRIVER_DELIVERIES DD ON DI.ITEM_ID = DD.ITEM_ID GROUP BY S.STAFF_ID, S.FIRST_NAME, S.SURNAME ORDER BY COUNT(*) DESC FETCH FIRST 1 ROW ONLY; DBMS_OUTPUT.PUT_LINE('--------------------------------------------'); DBMS_OUTPUT.PUT_LINE('STAFF ID: ' || v_staff_id); DBMS_OUTPUT.PUT_LINE('FIRST NAME: ' || v_first_name); DBMS_OUTPUT.PUT_LINE('SURNAME: ' || v_surname); DBMS_OUTPUT.PUT_LINE('DELIVERIES PROCESSED: ' || v_delivery_cnt); DBMS_OUTPUT.PUT_LINE('--------------------------------------------'); END; /
